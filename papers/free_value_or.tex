\begin{abstract}
  A set of free functions implementing the pattern of optional and expected
  \tcode{value_or} as free functions that produce the correct common reference
  type that was unfixable for \tcode{std::optional::value_or}.
\end{abstract}

% 4. Dangling is now detectable: reference_constructs_from_temporary_v

%
% 5. Uniform vocabulary across all pointer-like types: optional,
%    expected, raw pointers, smart pointers, optional<T&>. One
%    spelling for "the value, or this instead".

\section{Why free functions}

% Members cannot be retrofitted onto pointers, and changing
% optional::value_or's return type is an API/ABI break. A constrained
% free function template works uniformly and can compute the correct
% type.

\chapter{Implementation Experience}

% Reference implementation: beman.free_value_or
% \cite{Downey_free_value_or}. C++23 floor (needs
% reference_constructs_from_temporary_v). Tested across optional<T>,
% expected<T,E>, T*, shared_ptr<T>, unique_ptr<T>, and C++26
% optional<T&> (via beman.optional26), over engaged/disengaged,
% lvalue/rvalue/const value categories, constexpr evaluation, and
% negative-compile dangling cases.

\begin{minted}[fontsize=\small]{c++}
template <class T>
concept nullable = requires(const T t) {
    bool(t);
    *(t);
};

template <nullable T, class U,
          class R = std::common_reference_t<std::iter_reference_t<T>, U&&>>
constexpr auto reference_or(T&& m, U&& u) -> R;

template <nullable T, class U,
          class R = std::common_type_t<std::iter_reference_t<T>, U&&>>
constexpr auto value_or(T&& m, U&& u) -> R;

template <nullable T, class I,
          class R = std::common_type_t<std::iter_reference_t<T>,
                                       std::invoke_result_t<I>>>
constexpr auto or_invoke(T&& m, I&& invocable) -> R;
\end{minted}
